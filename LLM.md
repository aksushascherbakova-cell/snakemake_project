1. Создание основного скрипта
Создайте файл install_and_plot.sh со следующим содержимым:

#!/bin/bash

# Скрипт для установки R, необходимых пакетов и создания графика
# Требуется запуск с правами root

set -e  # Прерывать выполнение при любой ошибке
set -u  # Прерывать при использовании необъявленных переменных

# Определяем переменные
R_DIR="/root/python-r"
PLOT_SCRIPT="${R_DIR}/create_plot.R"
OUTPUT_FILE="${R_DIR}/ggplot2_plot.png"

# Функция для вывода информационных сообщений
log_info() {
    echo "[INFO] $(date '+%Y-%m-%d %H:%M:%S') - $1"
}

# Функция для вывода сообщений об ошибках
log_error() {
    echo "[ERROR] $(date '+%Y-%m-%d %H:%M:%S') - $1" >&2
    exit 1
}
